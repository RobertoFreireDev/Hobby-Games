function dump()
 printh("@@begin")
 for y=0,127 do
  local s=""
  for x=0,127 do
   local c=pget(x,y)
   s=s..sub("0123456789abcdef",c+1,c+1)
  end
  printh(s)
 end
 printh("@@end")
end
